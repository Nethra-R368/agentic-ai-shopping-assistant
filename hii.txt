# 1. Move into the backend folder
cd backend

# 2. (Optional but recommended) Activate your virtual environment if you use one
.\.venv\Scripts\Activate  

# 3. Start the backend server
uvicorn main:app --reload --host 0.0.0.0 --port 8000




# 1. Move into the frontend folder
cd frontend

# 2. Install dependencies (only needed if you haven't run this before or added new packages)
npm install

# 3. Start the frontend development server
npm run dev
